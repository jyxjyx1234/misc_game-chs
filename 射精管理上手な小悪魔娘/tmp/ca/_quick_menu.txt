






command $quick_menu_make_btn_sub( property $obj:object, property $file:str, property $xpos, property $ypos, property $btn_no, property $flag )
{
	$obj.create( $file, 1, $xpos, $ypos )
	$obj.set_button( $btn_no, 0, 24, 001 )
	if( $flag == 0 ){
		$obj.set_button_state_disable
	}
}




command $quick_menu_make_grp( property $anime_time )
{
	property $loop
	property $cont

	excall.back.objbtngroup[0].init

	
	excall.back.object[0].init
	excall.back.object[0].disp = 1
	excall.back.object[0].child.resize( 20 )

	excall.back.object[0].child[00].create(__sys_quick_00base,    1)		

	$quick_menu_make_btn_sub( excall.back.object[0].child[01], __sys_quick_01save,    525+    (210*0),  250+210*0,  0, syscom.check_save_enable)			
	$quick_menu_make_btn_sub( excall.back.object[0].child[02], __sys_quick_02load,    525+    (210*1),  250+210*0,  1, syscom.check_load_enable)			
	$quick_menu_make_btn_sub( excall.back.object[0].child[03], __sys_quick_04config,  525+    (210*2),  250+210*0,  2, 1)									
	$quick_menu_make_btn_sub( excall.back.object[0].child[04], __sys_quick_05hide,    525+    (210*3),  250+210*0,  3, syscom.check_hide_mwnd_enable)		
	$quick_menu_make_btn_sub( excall.back.object[0].child[05], __sys_quick_03backlog, 525+100+(210*0),  250+210*1,  4, syscom.check_msg_back_enable)		
	$quick_menu_make_btn_sub( excall.back.object[0].child[06], __sys_quick_11return,  525+100+(210*1),  250+210*1,  5, syscom.check_return_to_sel_enable)	
	$quick_menu_make_btn_sub( excall.back.object[0].child[07], __sys_quick_06auto,    525+100+(210*2),  250+210*1,  6, syscom.check_auto_mode_enable)		
	$quick_menu_make_btn_sub( excall.back.object[0].child[08], __sys_quick_07skip,    525+100+(210*3),  250+210*1,  7, syscom.check_read_skip_enable)		
	$quick_menu_make_btn_sub( excall.back.object[0].child[09], __sys_quick_08title,   525+200+(210*0),  250+210*2,  8, syscom.check_return_to_menu_enable)	
	$quick_menu_make_btn_sub( excall.back.object[0].child[10], __sys_quick_09exit,    525+200+(210*1),  250+210*2,  9, syscom.check_end_game_enable)		
	$quick_menu_make_btn_sub( excall.back.object[0].child[11], __sys_quick_10back,    525+200+(210*2),  250+210*2, 10, 1)									
	$quick_menu_make_btn_sub( excall.back.object[0].child[12], __sys_quick_12scenereturn,    525+200+(210*-1),  250+210*2, 11, $kaisou_mode != 0)									
	

	$quick_menu_make_checkbox( excall.back )

	wipe(0,$anime_time)
}

command $quick_menu_make_checkbox( property $stage : stage )
{
	if( syscom.get_auto_mode_onoff_flag ){	
		$stage.object[0].child[07].patno = 10
	}
	else{
		$stage.object[0].child[07].patno = 0
	}
	if( syscom.get_read_skip_onoff_flag ){	
		$stage.object[0].child[08].patno = 10
	}
	else{
		$stage.object[0].child[08].patno = 0
	}


}











command $quick_menu_main()
{
	property $mouse_click
	property $mouse_hit
	

	if( excall.check_alloc == 0 ){
		
		return( 0 )
	}

	
#initstart
	syscom.set_auto_mode_onoff_flag( 0 ) 	
	$quick_menu_make_grp( 200 )
	disp

#mainloop
	excall.front.objbtngroup[0].start_cancel
	input.clear
	while(1){
		$mouse_click = excall.front.objbtngroup[0].get_decided_no
		$mouse_hit = excall.front.objbtngroup[0].get_hit_no

		
		

		if (excall.front.object[0].child[01].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"セーブ画面を開きます。") 
		}
		elseif (excall.front.object[0].child[02].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"ロード画面を開きます。") 
		}
		elseif (excall.front.object[0].child[03].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"コンフィグを開きます。") 
		}
		elseif (excall.front.object[0].child[04].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"メッセージウィンドウを一時的に消去します。#D（スペースキーと同じ動作）") 
		}
		elseif (excall.front.object[0].child[05].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"メッセージバックログ（読み返し機能）を表示します。") 
		}
		elseif (excall.front.object[0].child[06].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"前の選択肢に戻ります。") 
		}
		elseif (excall.front.object[0].child[07].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"オートモードを開始します。") 
		}
		elseif (excall.front.object[0].child[08].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"文章スキップを開始します。") 
		}
		elseif (excall.front.object[0].child[09].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"タイトルに戻ります。") 
		}
		elseif (excall.front.object[0].child[10].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"ゲームを終了しWindowsに戻ります。") 
		}
		elseif (excall.front.object[0].child[11].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"前の画面に戻ります。") 	
		}
		elseif (excall.front.object[0].child[12].get_button_hit_state == 1){
			$help_text(excall.front.object[495],dr,"シーン回想画面にもどります。#D※シーン回想中のみ有効") 	
		}
		else{
			excall.front.object[495].disp = 0
		}




		
		
		if( $mouse_click == -2 ){	
			
		}
		elseif( $mouse_click == -1 ){	
			se.play(002)
			break
		}
		elseif( $mouse_click == 0 ){		
			
			se.play(001)
			syscom.call_save_menu
			break
		}
		elseif( $mouse_click == 1 ){		
			
			se.play(001)
			syscom.call_load_menu
			break
		}
		elseif( $mouse_click == 2 ){		
			se.play(001)
			
			$config_menu_main()
			goto #initstart
		}
		elseif( $mouse_click == 3 ){		
			se.play(001)
			script.set_mwnd_disp_off_flag(1)
			excall.front.object[0].disp = 0
			input.clear
			while(1){
				if( input.decide.on_down_up == 1 ){break}
				if( input.cancel.on_down_up == 1 ){break}
				input.next
				disp
			}
			script.set_mwnd_disp_off_flag(0)
			break
		}
		elseif( $mouse_click == 4 ){		
			se.play(001)
			$backlog_ready = 1
			break
		}
		elseif( $mouse_click == 5 ){		
			se.play(001)
			if( $dialog_main( excall.front, 2, excall.front.object[1], 3 ) ){
				syscom.return_to_sel( 0, 0, 1 )
				break
			}
			goto #mainloop
		}
		elseif( $mouse_click == 6 ){		
			se.play(001)
			if( syscom.get_auto_mode_onoff_flag == 0 ){
				syscom.set_auto_mode_onoff_flag( 1 ) 
			}
			else{
				syscom.set_auto_mode_onoff_flag( 0 ) 
			}
			$quick_menu_make_checkbox( excall.front )
			goto #mainloop
		}
		elseif( $mouse_click == 7 ){		
			se.play(001)
			syscom.set_read_skip_onoff_flag( 1 )
			break
		}
		elseif( $mouse_click == 8 ){		
			se.play(001)
			if( $dialog_main( excall.front, 2, excall.front.object[1], 4 ) ){
				if($kaisou_mode==0){
					syscom.end_save( 0, 0 )
				}
				syscom.return_to_menu( 0, 0, 1 )
			}
			goto #mainloop
		}
		elseif( $mouse_click == 9 ){		
			se.play(001)
			if( $dialog_main( excall.front, 2, excall.front.object[1], 6 ) ){
				
					se.stop
					for($se_tmp = 0,$se_tmp < 6,$se_tmp+=1) {$bgv_stop($se_tmp,500)}                 for($se_tmp = 0,$se_tmp < 3,$se_tmp+=1) {$bgs_oneshot_stop($se_tmp,500)}                 for($se_tmp = 0,$se_tmp < 3,$se_tmp+=1) {$bgs_action_stop($se_tmp,500)}                 for($se_tmp = 0,$se_tmp < 3,$se_tmp+=1) {$bgs_env_stop($se_tmp,500)}

					switch(math.rand(0,1)) { 					case(0)	 exkoe(012600038,001) 					case(1)	 exkoe(012600040,002) 					 				}
					if ($cr_mode == 1 && $cr_bgm != "") { } 		else {$now_bgm_name = "" bgm.stop(1000)}
					$all_screen("000000",1000,0,0,255,1)

					if (koe_check == 1) {
					
						koe_wait_key
					
					}            
					koe_stop

					syscom.end_game( 0, 0, 0 )
				

			}

			goto #mainloop
		}
		elseif( $mouse_click == 10 ){		
			se.play(001)
			break
		}

		elseif( $mouse_click == 11 ){		
			se.play(001)
			g[97] = 1
			syscom.return_to_menu( 0, 0, 0 )

			goto #mainloop
		}
		input.next
		disp
	}
	input.clear
	return
}







#z00

$backlog_ready = 0


if( excall.check_alloc == 0 ){
	capture
	excall.alloc
}

syscom.set_hide_mwnd_onoff_flag( 1 )

$quick_menu_main()

if( excall.check_alloc != 0 ){
	wipe( 0, 200 )
	excall.free
}
syscom.set_hide_mwnd_onoff_flag( 0 )

capture_free

if( $backlog_ready ){
	syscom.open_msg_back
}

return
