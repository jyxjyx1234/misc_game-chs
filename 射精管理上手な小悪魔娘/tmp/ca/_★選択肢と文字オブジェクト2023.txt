command	$obj_moji_centering(property $child,property $disp,property $x,property $y,property $color : str,property $fontsize,property $haikei_disp,property $haikei_color : str,property $moji : str) {

	front.object[150].child[$child].init
	front.object[150].child[$child].child.resize(2)
	front.object[150].child[$child].disp = $disp
	front.object[150].child[$child].x_rep.resize(2)
	front.object[150].child[$child].y_rep.resize(2)

	$ex_color_tmp2 = $color  		if ($ex_color_tmp2.len > 6) {   			$ex_color_tmp = $ex_color_tmp2.mid(0,3) $ex_color_r = $ex_color_tmp.tonum 			$ex_color_tmp = $ex_color_tmp2.mid(4,3) $ex_color_g = $ex_color_tmp.tonum 			$ex_color_tmp = $ex_color_tmp2.mid(8,3) $ex_color_b = $ex_color_tmp.tonum 			 		}else{  		 			$ex_color_r = $hexa_to_deci($ex_color_tmp2.mid(0,2)) 			$ex_color_g = $hexa_to_deci($ex_color_tmp2.mid(2,2)) 			$ex_color_b = $hexa_to_deci($ex_color_tmp2.mid(4,2)) 			 		 		}
	
	front.object[150].child[$child].child[1].create_string($moji,1,$x,$y)	
	front.object[150].child[$child].child[1].set_string_param($fontsize,0,0,0,0,0,0,0)
	front.object[150].child[$child].child[1].color_rate = 255
	front.object[150].child[$child].child[1].color_r = $ex_color_r
	front.object[150].child[$child].child[1].color_g = $ex_color_g
	front.object[150].child[$child].child[1].color_b = $ex_color_b
	
	l[00] = $moji.cnt
	l[01] = $moji.cnt * $fontsize
	
	
	
	if ($haikei_disp == 1) {
	
		$ex_color_tmp2 = $haikei_color  		if ($ex_color_tmp2.len > 6) {   			$ex_color_tmp = $ex_color_tmp2.mid(0,3) $ex_color_r = $ex_color_tmp.tonum 			$ex_color_tmp = $ex_color_tmp2.mid(4,3) $ex_color_g = $ex_color_tmp.tonum 			$ex_color_tmp = $ex_color_tmp2.mid(8,3) $ex_color_b = $ex_color_tmp.tonum 			 		}else{  		 			$ex_color_r = $hexa_to_deci($ex_color_tmp2.mid(0,2)) 			$ex_color_g = $hexa_to_deci($ex_color_tmp2.mid(2,2)) 			$ex_color_b = $hexa_to_deci($ex_color_tmp2.mid(4,2)) 			 		 		}
		front.object[150].child[$child].child[0].create_rect(-5,-5,l[01] + 5,$fontsize + 5,$ex_color_r,$ex_color_g,$ex_color_b,255,1,$x,$y)
	
	}
	
	
	
	
	front.object[150].child[$child].x_rep[0] = -((l[00] / 2) * $fontsize)
		
	if (l[00] % 2 == 1) {front.object[150].child[$child].x_rep[0] -= $fontsize / 2}	
	
}




command	$obj_moji_left_justified(property $child,property $disp,property $x,property $y,property $color : str,property $fontsize,property $haikei_disp,property $haikei_color : str,property $moji : str) {
	
	






	
	$moji_list.init
	$moji_tmp = ""
	$moji_list_count = 0
	

	if ($moji.search("R") == -1) {

		$moji_list[0] = $moji
		$moji_list_count = 1
		
	}else{
		
		$moji_list[0] = $moji.left($moji.search("R"))
		$moji_list[1] = $moji.mid($moji.search("R") + 1)
		
		$moji_list_count = 2
		
		if ($moji_list[1].search("R") != -1) {
			
			$moji_tmp = $moji_list[1]
			$moji_list[1] = $moji_list[1].left($moji_tmp.search("R"))
			$moji_list[2] = $moji_tmp.mid($moji_tmp.search("R") + 1)
			
			$moji_list_count = 3
		
		}
	
	}
		
	
	
	front.object[150].child[$child].init
	front.object[150].child[$child].child.resize(6)
	front.object[150].child[$child].disp = 1

	
	for(l[10] = 0,l[10] < $moji_list_count,l[10] += 1) {

		l[11] = l[10] * 2
		l[12] = ($fontsize + ($fontsize / 2)) * l[10]
		
		$ex_color_tmp2 = $color  		if ($ex_color_tmp2.len > 6) {   			$ex_color_tmp = $ex_color_tmp2.mid(0,3) $ex_color_r = $ex_color_tmp.tonum 			$ex_color_tmp = $ex_color_tmp2.mid(4,3) $ex_color_g = $ex_color_tmp.tonum 			$ex_color_tmp = $ex_color_tmp2.mid(8,3) $ex_color_b = $ex_color_tmp.tonum 			 		}else{  		 			$ex_color_r = $hexa_to_deci($ex_color_tmp2.mid(0,2)) 			$ex_color_g = $hexa_to_deci($ex_color_tmp2.mid(2,2)) 			$ex_color_b = $hexa_to_deci($ex_color_tmp2.mid(4,2)) 			 		 		}
		
		front.object[150].child[$child].child[l[11] + 1].create_string($moji_list[l[10]],1,$x,$y + l[12])	
		front.object[150].child[$child].child[l[11] + 1].set_string_param($fontsize,0,0,0,0,0,0,0)
		front.object[150].child[$child].child[l[11] + 1].color_rate = 255
		front.object[150].child[$child].child[l[11] + 1].color_r = $ex_color_r
		front.object[150].child[$child].child[l[11] + 1].color_g = $ex_color_g
		front.object[150].child[$child].child[l[11] + 1].color_b = $ex_color_b
		
		if(l[10] == 2) { front.object[150].child[$child].y -= ($fontsize + ($fontsize / 2)) }	
	
		l[00] = $moji_list[l[10]].cnt
		l[01] = $moji_list[l[10]].cnt * $fontsize
		
		
		
		if ($haikei_disp == 1) {
		
			$ex_color_tmp2 = $haikei_color  		if ($ex_color_tmp2.len > 6) {   			$ex_color_tmp = $ex_color_tmp2.mid(0,3) $ex_color_r = $ex_color_tmp.tonum 			$ex_color_tmp = $ex_color_tmp2.mid(4,3) $ex_color_g = $ex_color_tmp.tonum 			$ex_color_tmp = $ex_color_tmp2.mid(8,3) $ex_color_b = $ex_color_tmp.tonum 			 		}else{  		 			$ex_color_r = $hexa_to_deci($ex_color_tmp2.mid(0,2)) 			$ex_color_g = $hexa_to_deci($ex_color_tmp2.mid(2,2)) 			$ex_color_b = $hexa_to_deci($ex_color_tmp2.mid(4,2)) 			 		 		}
			front.object[150].child[$child].child[l[11] + 0].create_rect(-5,-5,l[01] + 5,$fontsize + 5,$ex_color_r,$ex_color_g,$ex_color_b,255,1,$x,$y + l[12])
		
		}
		
	}
	


}
