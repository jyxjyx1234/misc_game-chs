command $chr_emote(property $chrname:str,property $emote_no,property $se_play,property $se_wait,property $hanten_flag) {

    if ($chrname.len == 3) {    
        $chr_tansaku_hit_objno = -1 			$chr_tansaku_str = "" 			$chr_tansaku_chr = $chrname 			 			 			 			for($chr_tansaku_tmp = 20 ,$chr_tansaku_tmp <= 50 - 1,$chr_tansaku_tmp += 1) {	  				 				if (front.object[$chr_tansaku_tmp].exist_type == 1) { 				 						$chr_tansaku_str = front.object[$chr_tansaku_tmp].get_file_name 						 				 					if ($chr_tansaku_str.search($chr_tansaku_chr) != -1) { 					 						$chr_tansaku_hit_objno = $chr_tansaku_tmp 											 						break 					 					} 				 				} 			}
    }else{	
        $chr_tansaku_hit_objno = $chrname.tonum
    }

    if ($chr_tansaku_hit_objno == -1) {return}
    
    gosub #database


    front.object[$chr_tansaku_hit_objno].child[19].create_movie($chr_emote_database[$emote_no],0,0,0,ready_only = 1)
    front.object[$chr_tansaku_hit_objno].child[19].set_scale(1500,1500)
    front.object[$chr_tansaku_hit_objno].child[19].x_rep.resize(2)

    if ($hanten_flag == 0) {
        front.object[$chr_tansaku_hit_objno].child[19].x_rep[0] = -640
    }else{
        front.object[$chr_tansaku_hit_objno].child[19].x_rep[0] = 560
        front.object[$chr_tansaku_hit_objno].child[19].rotate_x = 1800
    }


    front.object[$chr_tansaku_hit_objno].child[19].y_rep.resize(2)
    front.object[$chr_tansaku_hit_objno].child[19].y_rep[0] = -2000

    switch($emote_no) { 
        case(2) front.object[$chr_tansaku_hit_objno].child[19].x_rep[0] = -480
        case(7) front.object[$chr_tansaku_hit_objno].child[19].x_rep[0] = -480
        case(8) front.object[$chr_tansaku_hit_objno].child[19].x_rep[0] = -600
        case(9) front.object[$chr_tansaku_hit_objno].child[19].x_rep[0] = -540
                front.object[$chr_tansaku_hit_objno].child[19].y_rep[0] = -2100
        case(10) front.object[$chr_tansaku_hit_objno].child[19].x_rep[0] = -480
                 front.object[$chr_tansaku_hit_objno].child[19].y_rep[0] = -2100

    }

    front.object[$chr_tansaku_hit_objno].child[19].disp = 1
    front.object[$chr_tansaku_hit_objno].child[19].resume_movie

    
    if ($se_play == 1) {
        pcmch[2].play($chr_emote_se_database[$emote_no])        
        if ($se_wait ==1) {
            pcmch[2].wait_key
        }
    }



}
